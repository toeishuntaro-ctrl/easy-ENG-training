export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } const { userText, targetPhrase, japaneseGuide } = req.body; const apiKey = process.env.GEMINI_API_KEY; // Vercelの環境変数から取得 const prompt = `You are an AI game engine for business English. Evaluate the user's response and output ONLY JSON. # CONTEXT - Target Key Phrase: "${targetPhrase}" - User Goal: "${japaneseGuide}" - User Spoken Response: "${userText}" # OUTPUT FORMAT (JSON) { "total_score": 85, "advice": "日本語での1行アドバイス", "ai_response_en": "Next AI response in English", "ai_response_jp": "次のAI応答の日本語訳", "next_target_phrase": "Next phrase", "next_japanese_guide": "Next guide" }`; try { const response = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: 'application/json' } }) } ); const data = await response.json(); const result = JSON.parse(data.candidates[0].content.parts[0].text); return res.status(200).json(result); } catch (error) { return res.status(500).json({ error: error.message }); } }
